function insertionSort(arr) {
  for (let i = 1; i < arr.length; i++) {
    const key = arr[i];   // the element to insert
    let j = i - 1;        // last index of the sorted part

    // shift larger elements one step to the right
    while (j >= 0 && arr[j] > key) {
      arr[j + 1] = arr[j];
      j--;
    }

    arr[j + 1] = key;     // drop the key into its place
  }
  return arr;
}

console.log(insertionSort([5, 3, 2, 4, 1])); // [1, 2, 3, 4, 5]
